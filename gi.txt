name:backend build and test  //name for this workflow
on:  // here we define the event that will trigger the workflow
  workflow_dispatch:  // this event will trigger the workflow manually
                  
                  (or)
   
  push: // push event trigger
    branches:  // trigger on push to main branch
      - main  // main branch


env: //reuseable variable
  DOCKER_IMAGE_NAME: kavi8667/backend:sass // docker image name


jobs:   // here we create the jobs for the workflow mean (build,test,docker file create ,docker run and deploy)
   build-and-test-job: //name of the job
    runs-on: ubuntu-latest // specify the operating system for the job

    steps: // here we define the steps for the job
      - name: code checkout pannu  // first checkout step for code 
        uses: actions/checkout@v7  // use checkout the code from the repository privately

      - name: Setup Node.js  // setup nodejs environment for the build
        uses: actions/setup-node@v7 // use setup-node action to configure Node.js environment
        with:
          node-version: 20  // specify the Node.js version to use for the build

      - name: Install dependencies // install the project dependencies using npm
        run: npm i  // run npm ci to install dependencies from package-lock.json

      - name: Build // build the project using npm run build command
        run: npm run build // run the build script defined in package.json to compile the project
              - name : docker image build pannu  //docker image build pannu
        run : docker build -t kavi8667/backend:github .  //docker image build tags docker hub

      - name: Login to Docker Hub  //docker hub login pannu
        uses: docker/login-action@v4  //docker hub login action
        with:
          username: ${{vars.DOCKER_USER_NAME}}  //docker hub username
          password: ${{secrets.DOCKER_PASS}}  //docker hub password

      - name: Push Docker image to Docker Hub  //docker image push pannu
        run: docker push kavi8667/backend:github  //docker image push to docker hub

  
  deploy-job:  //next job deploy
    needs: build-and-test-job  // this wait for complete the assigned job then execute
    runs-on: ubuntu-latest //run one vm meachine

    steps:
      -name: server ssh  login panna // name for ssh login panna
      uses: appleboy/ssh-action@v1  // https://github.com/appleboy/ssh-action
        with:
          host: ${{vars.HOST_IP}}  // repo-->setting-->action--> public ip
          username: ec-user // user name
          key: ${{ secrets.AWS_EC2_KEY }}  //  aws instanve ppk key same  get repo-->setting-->action-->secrets-->secrets_key
          port: 22 //ssh port
          script: |
            sudo apt-get update
            sudo apt-get install -y docker.io

            sudo systemctl start docker
            sudo systemctl enable docker

            sudo docker --version
            sudo systemctl status docker --no-pager
            sudo docker rmi ${{env.DOCKER_IMAGE_NAME}}
            sudo docker pull ${{ env.DOCKER_IMAGE_NAME }}:sass
            sudo docker stop sass-backend || true
            sudo docker rm sass-backend || true
            sudo docker run -d \
            --name sass-backend \
            -p 3000:3000 \
            ${{ env.DOCKER_IMAGE_NAME }}:sass
