name:sass build and deploy
on:
  push:
    branches:
      - main

env: //reuseable variable
  DOCKER_IMAGE_NAME: kavi8667/backend:sass // docker image name


jobs:
  build-and-test-job:
    env:
      summa: simple // it use in this job only
    runs-on: ubuntu-latest
    steps:
      -name: checkout code
       uses: actions/checkout@v2
       env:
         summa1: simple // it use in this step only

      -name: set node
       uses: actions/setup-node@v2
       with: 
         node-version: 20
      - name: install dependencies
       run: npm install

       -name: build sass
       run: npm run build

       -name: build docker image
       run: docker build -t ${{env.DOCKER_IMAGE_NAME}} .

       -name: login to docker hub
       uses:docker/login-action@v2
       with:
        username:${{secrets.DOCKER_USERNAME}}
        password:${{secrets.DOCKER_PASSWORD}}

      -name: push docker image
       run: docker push ${{env.DOCKER_IMAGE_NAME}}
  
  
  deploy-job:  //next job deploy
    needs: build-and-test-job  // this wait for complete the assigned job then execute
    runs-on: ubuntu-latest //run one vm meachine

    steps:
      -name: server ssh  login panna // name for ssh login panna
      uses: appleboy/ssh-action@v1  // https://github.com/appleboy/ssh-action
        with:
          host: ${{vars.HOST_IP}}  // repo-->setting-->action--> public ip
          username: ec-user // user name
          key: ${{ secrets.AWS_EC2_KEY }}  //  aws instanve ppk key same  get repo-->setting-->action-->secrets-->secrets_key
          port: 22 //ssh port
          script: |
            sudo apt-get update
            sudo apt-get install -y docker.io

            sudo systemctl start docker
            sudo systemctl enable docker

            sudo docker --version
            sudo systemctl status docker --no-pager
            sudo docker rmi ${{env.DOCKER_IMAGE_NAME}}
            sudo docker pull ${{ env.DOCKER_IMAGE_NAME }}:sass
            sudo docker stop sass-backend || true
            sudo docker rm sass-backend || true
            sudo docker run -d \
            --name sass-backend \
            -p 3000:3000 \
            ${{ env.DOCKER_IMAGE_NAME }}:sass
